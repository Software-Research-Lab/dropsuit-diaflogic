/**
 * @description Generates a Dialogflow message bubble and buttons.
 * @param {object} capsule - The object returned by the capsule_f function containing CSS design classes,
 * button names, and button condition update function.
 * @param {string} message - The agent Dialogflow message.
 * @param {array} btns - An array of button names.
 * @param {boolean} dispout - Whether to display processing output results in terminal.
 * @returns {object} - An object containing the agent Dialogflow message HTML bubble and buttons.
 * @example
 * function slot_f(capsule, message, btns, dispout) {}
 */

function slot_f(capsule, message, btns, dispout) {
  if (btns == null) {
    let bubbleMess = bubble_f(capsule, message, false);
    const bubMessBtn = {
      btn: "",
      message: bubbleMess,
    };
    return bubMessBtn;
  } else {
    capsule.btnFlow = btns; // capsule.btnFlow = null;
    const btnFnc = btnHTML_ClassFuncName(capsule.dbf.btnCN, capsule.dbf.funcN);
    const httpLink = httpHTML_LinkName(capsule.dbf.btnCN);
    let btn,
      buttons = "";
    let cap = capsule.btnFlow;
    for (c = 0; c < cap.length; c++) {
      let btnName = cap[c];
      let res = htmlText(btnName);
      if (res == false) {
        btn = btnHTML_CFN_NameValue(btnFnc, btnName, btnName);
        buttons = buttons + btn;
      } else if (res == true) {
        btn = btnHTML_Link(httpLink, btnName);
        buttons = buttons + btn;
      }
    }
    const btnGroup = btnHTML_CFNNV_divClassBtn(capsule.dbf.divCN, buttons);
    let bubbleMess = bubble_f(capsule, message, false);
    let combBtnMess = blow_f(btnGroup, bubbleMess);
    displayP2(combBtnMess, dispout); /// DISPLAY >>
    return combBtnMess;
  }
}

function blow_f(btnGroup, bubbleMess) {
  const bubMessBtn = {
    btn: btnGroup,
    message: bubbleMess,
  };
  return bubMessBtn;
}

function htmlText(htmlTxt) {
  if (htmlTxt.indexOf("http://") == 0 || htmlTxt.indexOf("https://") == 0) {
    return true;
  } else {
    return false;
  }
}

// link

function httpHTML_LinkName(btnClassName) {
  const link = {
    alink: "<p class='" + btnClassName + "'> <a href='",
    alinkend: "' target='_blank'>",
    btnend: "</a></p>",
  };
  return link;
}

function btnHTML_Link(buildBtn, btnName) {
  var namelink = btnName.split(/[>]/);
  //console.log("\n\nSPLIT RES:", namelink, "\n\n");
  let oneLink =
    buildBtn.alink +
    namelink[0] +
    buildBtn.alinkend +
    namelink[1] +
    buildBtn.btnend;
  return oneLink;
}

// btn

function btnHTML_ClassFuncName(btnClassName, functionName) {
  const buildBtn = {
    btndiv:
      "<button class='" + btnClassName + "' onclick='" + functionName + "(",
    fncend: ")'>",
    btnend: "</button>",
  };
  return buildBtn;
}

function btnHTML_CFN_NameValue(buildBtn, btnName, btnValue) {
  let oneBtn =
    buildBtn.btndiv +
    '"' +
    btnValue +
    '"' +
    buildBtn.fncend +
    btnName +
    buildBtn.btnend;
  return oneBtn;
}

function btnHTML_CFNNV_divClassBtn(btnsClassName, button_s) {
  const buildBtn = {
    div: "<div class='" + btnsClassName + "'>",
    divend: "</div>",
  };
  let btnGroup = buildBtn.div + button_s + buildBtn.divend;
  return btnGroup; /// "<div class='buttons'> <button class='btn' onclick='sendButton(1)'>YES</button> </div>"
}

function agent_f(input, defAgent, agents, dispout) {
  var inp = ds_clnstr.clnstr(input).txt();
  let istrue = [];
  if (agents.includes(inp)) {
    defAgent = inp;
    displayR1(defAgent, dispout); /// DISPLAY >>>
    return defAgent;
  } else {
    let insub, subin, extractAgent;
    for (a = 0; a < agents.length; a++) {
      insub = sub(inp, agents[a]);
      subin = sub(agents[a], inp);
      if (insub == true || subin == true) {
        console.log("\n\nINSUB:", insub, "\n\nSUBIN:", subin);
        if (!istrue.includes(true)) {
          extractAgent = agents[a];
          istrue.push(true);
        }
      }
    }
    if (istrue.includes(true)) {
      console.log("\n\nINCLUDES TRUE!");
      defAgent = inp;
      displayR1(defAgent, dispout); /// DISPLAY >>>
      return (defAgent = extractAgent);
    } else {
      displayR1(defAgent, dispout); /// DISPLAY >>>
      return defAgent;
    }
  }
}

function sub(inpA, inpB) {
  if (inpA.indexOf(inpB) !== -1) {
    return true;
  } else if (inpA.includes(inpB)) {
    return true;
  } else if (inpA.search(inpB) !== -1) {
    return true;
  } else if (inpA.match(inpB) !== null) {
    return true;
  } else {
    return false;
  }
}

/**
 * @description Encapsulates bot output message bubble CSS design classes,
 * button names, and button condition update function.
 * @param {object} designObj - The .json loader output object that returns CSS design classes.
 * @param {string} input - The input text.
 * @param {string} fncValue - The function value name.
 * @param {boolean} dispout - Whether to display processing output results in terminal.
 * @returns {object} - An object containing CSS design classes, button names, and button condition update function.
 * @example
 * function capsule_f(designObj, input, fncValue, dispout) {}
 */

function capsule_f(designObj, input, fncValue, dispout) {
  const can = {
    funcN: designObj.funcname,
    divCN: designObj.divclass,
    btnCN: designObj.btnclass,
    bublC: designObj.bubbleclass,
  };
  const capsule = {
    dbf: can,
    cond: "",
    btnFlow: [],
  };
  let condMess = dialogflowGate(fncValue, capsule);
  capsule.cond = condMess;
  displayP1(input, fncValue, dispout, capsule); /// DISPLAY >>
  return capsule;
}

function dialogflowGate(fncValue, capsule) {
  if (fncValue !== undefined && fncValue !== "undefined") {
    fncValue = ds_clnstr.clnstr(fncValue).txt();
    return fncValue;
  } else {
    let btns = [];
    slot_f(capsule, null, btns);
    return false;
  }
}

/**
 * @description Handles undefined conditions of Dialogflow results, including out-of-thread value,
 * and configures text message HTML bubble with buttons.
 * @param {string} outMessFlowConv - A separate 'out of Dialogflow Message' nsent(null/string, integer, integer, integer)
 * function answer search result to the input text message request.
 * @param {object} capsule_txDlgMess - The default message design content.
 * @param {object} capsule_NoAnsMess - The 'No answer' condition message design content.
 * @param {string} slotMessage - The agent Dialogflow message.
 * @param {object} btnGroup - An object containing the agent Dialogflow message HTML bubble and buttons.
 * @param {boolean} dispout - Whether to display processing output results in terminal.
 * @returns {object} - An object containing the agent Dialogflow message HTML bubble and buttons.
 * @example
 * function flow_f(outMessFlowConv, capsule_txDlgMess, capsule_NoAnsMess, slotMessage, btnGroup, dispout) {}
 */

function flow_f(
  outMessFlowConv,
  capsule_txDlgMess,
  capsule_NoAnsMess,
  slotMessage,
  btnGroup,
  dispout
) {
  if (slotMessage == undefined || slotMessage == "undefined") {
    let cap;
    if (outMessFlowConv == false) {
      cap = capsule_NoAnsMess;
      outMessFlowConv = "Lack of data... ~00^";
    } else {
      cap = capsule_txDlgMess;
    }
    let bubbleMess = bubble_f(cap, outMessFlowConv, false);
    if (btnGroup != undefined) {
      btnGroup.message = bubbleMess;
    } else {
      const undefinedBtn = {
        btn: "",
        message: bubbleMess,
      };
      btnGroup = undefinedBtn;
    }
    displayP3(outMessFlowConv, bubbleMess, dispout); /// DISPLAY >>
    return btnGroup;
  } else {
    displayP3(outMessFlowConv, null, dispout); /// DISPLAY >>
    return btnGroup;
  }
}

let boolSet = [];

/**
 boolset_f
* @description Sets the actual agent condition.
* @param {Array} otheragents - An array of agent names.
* @param {string} defagent - The actual agent name.
* @param {boolean} agentbool - The actual condition of the agent.
* @param {boolean} dispout - (true/false) Display processing output results in terminal.
* @returns {string} - The actual agent condition.
 */

function boolset_f(otheragents, defagent, agentbool, dispout) {
  boolSet = [];
  for (a = 0; a < otheragents.length; a++) {
    boolSet.push(false);
  }
  if (otheragents.includes(defagent)) {
    let index = otheragents.indexOf(defagent);
    boolSet[index] = agentbool;
  }
  let index = boolSet.indexOf(true);
  let acualAgent = otheragents[index];
  return acualAgent;
}
